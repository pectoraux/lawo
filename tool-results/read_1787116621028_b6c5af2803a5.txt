     1→# Nomos — Universal Rules-and-Reality Operating System
     2→
     3→## Project Identity
     4→
     5→- **Name**: Nomos (from Greek: law / custom / order)
     6→- **Tagline**: A universal rules-and-reality operating system
     7→- **Stack**: Next.js 16 + TypeScript 5 + Tailwind 4 + shadcn/ui + Prisma (SQLite) + Zustand
     8→- **Single user-visible route**: `/`
     9→
    10→## Frozen Architecture (see architecture/)
    11→
    12→The platform is built on a frozen architecture with 5 planes:
    13→- A. Experience Plane (consumer UI)
    14→- B. Intelligence Plane (context/state/rule/decision engines)
    15→- C. Knowledge Plane (entity/fact/jurisdiction/rule graphs)
    16→- D. Execution Plane (integrations/actions)
    17→- E. Platform Foundation (tenancy/identity/audit)
    18→
    19→The kernel is **domain-agnostic**. Vertical behavior lives in packages.
    20→

    25→### Kernel primitives (`src/kernel/primitives/types.ts`)
    26→
    27→```typescript
    28→// ===== Truth / Confidence Model (T0–T5) =====
    29→export type TruthLevel = 'T0' | 'T1' | 'T2' | 'T3' | 'T4' | 'T5';
    30→// T0 authoritative, T1 deterministically derived, T2 established interpretation,
    31→// T3 expert interpretation, T4 community observation, T5 prediction
    32→
    33→export type RuleType = 'DETERMINISTIC' | 'CONDITIONAL' | 'DISCRETIONARY' | 'PREDICTIVE';
    34→
    35→export type ObservationStatus = 'OFFICIAL' | 'VERIFIED' | 'COMMUNITY_REPORTED' | 'UNVERIFIED' | 'PREDICTED';
    36→
    37→// ===== Temporal Model =====
    38→export interface TemporalRange {
    39→  validFrom: string;       // ISO date
    40→  validTo?: string | null; // ISO date or null = open-ended
    41→  publishedAt?: string;
    42→  ingestedAt?: string;
    43→  version: number;
    44→  supersedes?: string | null;
    45→  supersededBy?: string | null;
    46→}
    47→
    48→// ===== Provenance =====
    49→export interface Provenance {
    50→  decisionId: string;
    51→  ruleId: string;
    52→  ruleVersion: number;
    53→  source: SourceRef;
    54→  authority: AuthorityRef;
    55→  facts: FactRef[];
    56→  evidence: EvidenceRef[];
    57→  calculation: CalculationStep[];
    58→  assumptions: string[];
    59→  truthLevel: TruthLevel;
    60→  asOf: string;             // ISO date — the "evaluate as_of" anchor
    61→  producedAt: string;       // ISO timestamp
    62→}
    63→
    64→export interface SourceRef { sourceId: string; citation: string; url?: string; }
    65→export interface AuthorityRef { authorityId: string; name: string; jurisdictionId: string; }
    66→export interface FactRef { factId: string; subjectId: string; attribute: string; value: unknown; truthLevel: TruthLevel; }
    67→export interface EvidenceRef { evidenceId: string; documentId?: string; page?: number; region?: string; }
    68→export interface CalculationStep { description: string; input: unknown; output: unknown; ruleClause?: string; }
    69→
    70→// ===== Kernel Primitives =====
    71→export interface Entity { id: string; type: string; label: string; tenantId: string | null; attributes?: Record<string, unknown>; }
    72→export interface Fact {
    73→  id: string; subjectId: string; attribute: string; value: unknown;
    74→  truthLevel: TruthLevel; source?: SourceRef; observedAt: string;
    75→  tenantId: string | null; jurisdictionId?: string;
    76→}
    77→export interface Jurisdiction {
    78→  id: string; code: string; name: string; kind: 'COUNTRY' | 'REGION' | 'STATE' | 'MUNICIPALITY' | 'REGULATOR' | 'COURT' | 'SPECIAL_ZONE' | 'FREE_ZONE' | 'SUPRANATIONAL' | 'BILATERAL' | 'INTERNATIONAL';
    79→  parentIds: string[]; temporal: TemporalRange;
    80→}
    81→export interface JurisdictionEdge { fromId: string; toId: string; relation: JurisdictionRelation; }
    82→export type JurisdictionRelation = 'APPLIES_TO' | 'OVERRIDES' | 'PREEMPTS' | 'IMPLEMENTS' | 'DERIVES_FROM' | 'MODIFIES' | 'EXEMPTS' | 'REFERENCES' | 'SUPERSEDES' | 'INTERPRETS' | 'CONDITIONAL_ON';
    83→export interface Authority { id: string; name: string; jurisdictionId: string; kind: 'LEGISLATURE' | 'EXECUTIVE' | 'JUDICIARY' | 'REGULATOR' | 'INTERNATIONAL_BODY' | 'CUSTOMS' | 'IMMIGRATION' | 'TAX' | 'OTHER'; }
    84→export interface Source { id: string; title: string; citation: string; url?: string; authorityId: string; publishedAt?: string; }
    85→export interface Rule {
    86→  id: string; code: string; title: string; jurisdictionId: string; authorityId: string; sourceId: string;
    87→  type: RuleType; ruleIr: RuleIR; temporal: TemporalRange; packageId: string; truthLevel: TruthLevel;
    88→}
    89→export interface RuleIR {
    90→  id: string; ruleId: string;
    91→  conditions: ConditionNode;          // boolean expression tree over facts
    92→  exceptions: ConditionNode[];         // if any true, rule does not apply
    93→  effects: RuleEffect[];              // rights/obligations/permissions/restrictions granted/denied
    94→  definitions?: Record<string, Definition>;
    95→  references?: string[];              // sourceId refs
    96→  interpretiveStatus?: 'SETTLED' | 'CONTESTED' | 'AMBIGUOUS';
    97→}
    98→export type ConditionNode =
    99→  | { kind: 'leaf'; fact: string; operator: 'eq' | 'neq' | 'gt' | 'gte' | 'lt' | 'lte' | 'in' | 'contains' | 'exists'; value: unknown }
   100→  | { kind: 'and'; children: ConditionNode[] }
   101→  | { kind: 'or'; children: ConditionNode[] }
   102→  | { kind: 'not'; child: ConditionNode };
   103→export type EffectKind = 'RIGHT' | 'OBLIGATION' | 'PERMISSION' | 'RESTRICTION' | 'FEE' | 'OPTION' | 'CONSEQUENCE';
   104→export interface RuleEffect { kind: EffectKind; code: string; label: string; detail?: string; amount?: { value: number; currency: string; basis?: string }; }
   105→export interface Definition { term: string; meaning: string; }
   106→
   107→// ===== State =====
   108→export interface StateSnapshot {
   109→  situationId: string; subjectId: string; jurisdictionIds: string[];
   110→  asOf: string; computedAt: string;
   111→  applicableRules: Rule[]; firedEffects: FiredEffect[]; options: Option[]; obligations: Obligation[]; rights: Right[]; permissions: Permission[]; restrictions: Restriction[];
   112→  truthLevel: TruthLevel;
   113→  provenance: Provenance[];
   114→}
   115→export interface FiredEffect { ruleId: string; effect: RuleEffect; truthLevel: TruthLevel; }
   116→export interface Option { id: string; code: string; label: string; detail?: string; preconditions?: ConditionNode; actionId?: string; }
   117→export interface Obligation { id: string; code: string; label: string; dueBy?: string; authorityId: string; }
   118→export interface Right { id: string; code: string; label: string; }
   119→export interface Permission { id: string; code: string; label: string; }
   120→export interface Restriction { id: string; code: string; label: string; }
   121→
   122→// ===== Situations & Procedures =====
   123→export interface SituationState { id: string; label: string; description?: string; isTerminal?: boolean; }
   124→export interface SituationTransition { from: string; to: string; event: string; requiredFacts?: string[]; preconditions?: ConditionNode; }
   125→export interface Situation {
   126→  id: string; code: string; label: string; description: string; packageId: string;
   127→  entryConditions: ConditionNode; states: SituationState[]; transitions: SituationTransition[];
   128→  requiredFacts: string[]; applicableDomains: string[]; actors: string[];
   129→  procedures: string[]; possibleActions: string[]; exitConditions: ConditionNode; exceptionPaths?: string[];
   130→}
   131→export interface ProcedureStep {
   132→  id: string; code: string; label: string; description?: string;
   133→  requiredDocuments?: string[]; acceptedAlternatives?: string[]; expectedOutputs?: string[];
   134→  fees?: { label: string; amount: number; currency: string }[]; timing?: string; nextStep?: string; exceptionPath?: string;
   135→}
   136→export interface Procedure { id: string; code: string; label: string; situationId: string; steps: ProcedureStep[]; }
   137→
   138→// ===== Actions =====
   139→export interface Action {
   140→  id: string; code: string; label: string; description?: string; kind: 'FILE' | 'PAY' | 'NOTIFY' | 'NAVIGATE' | 'SUBMIT' | 'GENERATE_DOCUMENT' | 'REQUEST_INFO' | 'REPORT' | 'HANDOFF';
   141→  preconditions?: ConditionNode; executionHint?: string; expectedResult?: string;
   142→}
   143→
   144→// ===== Packages =====
   145→export type PackageCategory = 'JURISDICTION' | 'DOMAIN' | 'SITUATION' | 'CAPABILITY';
   146→export interface PackageManifest {
   147→  packageId: string; name: string; version: string; category: PackageCategory;
   148→  dependencies: { packageId: string; versionRange: string }[];
   149→  supportedJurisdictions: string[]; domains: string[]; situations: string[]; capabilities: string[];
   150→  sources: string[]; rules: string[]; procedures: string[]; actions: string[]; schemas: string[];
   151→  testFixtures: string[]; verificationMetadata: { signedBy: string; signedAt: string; hash: string };
   152→  description: string;
   153→}
   154→
   155→// ===== Evidence =====
   156→export interface Document { id: string; type: string; title: string; tenantId: string | null; pages?: number; }
   157→export interface Evidence { id: string; documentId?: string; page?: number; region?: string; extractedFactIds: string[]; confidence: number; }
   158→
   159→// ===== Context =====
   160→export interface ContextRequest {
   161→  subjectId: string; locationId?: string; asOf: string; situationId?: string;
   162→  facts: Fact[]; jurisdictionIds: string[]; objective?: string; tenantId: string | null;
   163→}
   164→export interface ContextBundle {
   165→  request: ContextRequest; resolvedJurisdictions: Jurisdiction[]; resolvedAuthorities: Authority[];
   166→  applicableRules: Rule[]; evidence: Evidence[]; sources: Source[];
   167→}
   168→
   169→// ===== Audit =====
   170→export interface AuditEvent {
   171→  id: string; tenantId: string | null; actor: string; action: string; subjectId?: string;
   172→  timestamp: string; severity: 'INFO' | 'WARN' | 'ERROR' | 'CRITICAL'; payload: Record<string, unknown>;
   173→}
   174→
   175→// ===== Tenant =====
   176→export interface Tenant { id: string; name: string; kind: 'INDIVIDUAL' | 'HOUSEHOLD' | 'SMALL_BUSINESS' | 'ENTERPRISE' | 'PROFESSIONAL_ORG' | 'GOVERNMENT' | 'EMBEDDED'; createdAt: string; }
   177→```
